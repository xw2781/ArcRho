Example

Send progress from Python

from arcrho_api import ArcRhoUI

ui = ArcRhoUI()
ui.wait_for_app(timeout_sec=10)

tasks = ui.task_designer()
tasks.open(title="Import from ResQ")
tasks.set_tasks([
    {"name": "Connect", "description": "Open ResQ COM session"},
    {"name": "Export", "description": "Write ArcRho files"},
])
tasks.start_task("Connect")
tasks.complete_task("Connect", comments="Connected.")
tasks.start_task("Export")
tasks.update_task("Export", comments="Writing DFM methods...")
